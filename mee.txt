RunService.RenderStepped:Connect(function(dt)
    if not espEnabled then
        for _, data in pairs(espLines) do data.line.Visible = false data.text.Visible = false end
        if espOpahHighlight then espOpahHighlight.FillTransparency = 1 espOpahHighlight.OutlineTransparency = 1 end

        return

    local viewport = Camera.ViewportSize
    local originX = viewport.X / 2
    local originY = 0
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    for _, data in pairs(espLines) do
        local part = data.part
        if part and part.Parent then
            local distance = hrp and (hrp.Position - part.Position).Magnitude or 0
            local maxDistance = 500 
            local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
            if onScreen and screenPos.Z > 0 and distance <= maxDistance then
                if data.alpha < 1 then data.alpha = math.min(1, data.alpha + dt * data.fadeSpeed) end
                data.line.Color = rainbowColor
                data.text.Color = rainbowColor
                data.line.Transparency = data.alpha
                data.text.Transparency = data.alpha
                data.line.Visible = true
                data.line.From = Vector2.new(originX, originY)
                data.line.To = Vector2.new(screenPos.X, screenPos.Y)
                data.text.Visible = true
                data.text.Position = Vector2.new(screenPos.X, screenPos.Y - 15)
            else
                data.line.Visible = false
                data.text.Visible = false
                data.alpha = 0 
            end
        else
            data.line.Visible = false
            data.text.Visible = false
        end
    end
    if espOpahHighlight and espOpahBillboard then
        local opah = espOpahHighlight.Adornee
        if opah and opah.Parent then
            local opahPos = opah:GetPivot().Position
            local distance = hrp and (hrp.Position - opahPos).Magnitude or 0
            if distance <= 500 then
                espOpahHighlight.FillTransparency = math.max(0.5, espOpahHighlight.FillTransparency - dt * 1.5)
                espOpahHighlight.OutlineTransparency = math.max(0, espOpahHighlight.OutlineTransparency - dt * 1.5)
                espOpahBillboard.Enabled = true
            else
                espOpahHighlight.FillTransparency = 1
                espOpahHighlight.OutlineTransparency = 1
                espOpahBillboard.Enabled = false
            end
        end
    end
end)
